437  /*Five Top function*/
438  void fiveTop() {
439      ifstream file("student.txt", ios::in);
440  
441  
442  
443  
444            if (file)
445      {
446  
447          /*Five Top variables Help Me With Array*/
448          float top[100]{};
449          int sum{};
450          float avg{};
451          int count{};
452          int largeIndex{};
453  
454          /*Get The Average for each student and add it to the array*/
455          while (file >> student.id >> student.first_name >> student.last_name >> student.gender >> student.phone_number >> student.department)
456          {
457              for (size_t i = 0; i < 7; i++)
458              {
459                  file >> student.marks[i];
460                  sum += student.marks[i];
461              }
462              top[count] = sum / 7.0;
463              sum = 0;
464              count++;
465          }
466  
467  
468          /*Sort The array with the five top*/
469          for (size_t i = 0; i < count - 1; i++)
470          {
471              largeIndex = i;
472              for (size_t j = i + 1; j < count; j++)
473              {
474                  if (top[largeIndex] < top[j]) {
475                      largeIndex = j;
476                  }
477              }
478  
479              float temp = top[largeIndex];
480              top[largeIndex] = top[i];
481              top[i] = temp;
482          }
483  
484          /*Print The Five Top Students*/
485          ifstream file("student.txt", ios::in);
486          cout << "The First Five Top Students Are: " << endl;
487          while (file >> student.id >> student.first_name >> student.last_name >> student.gender >> student.phone_number >> student.department)
488          {
489              for (size_t i = 0; i < 7; i++)
490              {
491                  file >> student.marks[i];
492                  sum += student.marks[i];
493              }
494              avg = sum / 7.0;
495  
496              if (avg == top[0] || avg == top[1] || avg == top[2] || avg == top[3] || avg == top[4]) {
497                  cout << "ID: " << student.id
498                      << "\nFirst name: " << student.first_name
499                      << "\nLast name: " << student.last_name
500                      << "\nAVG: " << avg << endl << endl;
501              }
502  
503              sum = 0;
504              avg = 0;
505  
506          }
507          file.close();
508      }
509      else
510      {
511          cout << "We can't open this file....\n";
512      }
